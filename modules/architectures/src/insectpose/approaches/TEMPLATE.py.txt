from __future__ import annotations

from pathlib import Path
from typing import Any

import pandas as pd

from insectpose.approaches.base import BaseApproach
from insectpose.context import RunContext
from insectpose.data.datamodule import FoldData, ImageSet
from insectpose.registry import register_approach


@register_approach("APPROACH_NAME")  # must be identical to the name of the YAML
class TemplateApproach(BaseApproach):
    """One sentence: what this approach tests and what it assumes."""

    def fit(self, data: FoldData, ctx: RunContext) -> None:
        """Train on data.train, validate on data.val.

        RULES:
        - never read data.test (a unit test checks it);
        - write only into ctx.subdir(...);
        - seeds through ctx.seed('...');
        - the framework metrics are for monitoring, never for comparison;
        - fill the costs in ctx.extra (model_params, train_time_s, peak_vram_mb).
        """
        raise NotImplementedError

    def predict_instances(self, images: ImageSet, ctx: RunContext) -> pd.DataFrame:
        """Return the predictions IN THE FRAME OF THE ORIGINAL IMAGE.

        Columns: image_id, bbox_xywh, bbox_score, kpts_xy, kpts_score,
        keypoint_schema, bbox_source, inference_ms (optional).

        PITFALLS:
        - work on a crop -> back-project with invert_affine (§9.3);
        - multi-dataset model -> reproject from the union space to the LOCAL schema;
        - do NOT threshold the scores here: thresholding is an evaluation operation.
        """
        raise NotImplementedError

    @classmethod
    def load(cls, run_dir: Path, cfg: Any) -> TemplateApproach:
        """Reload the model from run_dir/weights without retraining."""
        raise NotImplementedError
